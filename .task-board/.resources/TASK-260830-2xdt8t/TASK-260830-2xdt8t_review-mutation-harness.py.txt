def restore():
    for path, orig in ORIG.items():
        shutil.copyfile(orig, path)

results = []
for name, path, old, new, pkg, run in MUTANTS:
    restore()
    src = open(path).read()
    n = src.count(old)
    if n != 1:
        results.append((name, "MUTANT-NOT-APPLIED(count=%d)" % n, ""))
        continue
    open(path, "w").write(src.replace(old, new, 1))
    p = subprocess.run(["go", "test", pkg, "-run", run, "-count=1"],
                       cwd=W, capture_output=True, text=True)
    if p.returncode != 0 and ("build failed" in p.stdout + p.stderr or "cannot" in p.stdout+p.stderr and "FAIL" not in p.stdout):
        status = "BUILD-ERROR"
    elif p.returncode != 0:
        status = "KILLED (tests red)"
    else:
        status = "SURVIVED (tests still green)"
    results.append((name, status, (p.stdout + p.stderr).strip().splitlines()[-3:] if p.returncode != 0 else []))
restore()

for name, status, tail in results:
    print("%-62s %s" % (name, status))
open(os.path.join(D, "mutation-results.json"), "w").write(json.dumps([{"mutant": n, "status": s} for n, s, _ in results], indent=2))

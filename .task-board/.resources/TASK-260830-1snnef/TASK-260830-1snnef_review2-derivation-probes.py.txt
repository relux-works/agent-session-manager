import hashlib, json, os, subprocess, sys

ROOT = os.getcwd()
PKG = "internal/terminalbackend"
FILES = [os.path.join(PKG, f) for f in os.listdir(PKG) if f.endswith(".go")] + [
    "internal/config/validation.go"]
ORIG = {f: open(f).read() for f in FILES}
SUMS = {f: hashlib.sha256(v.encode()).hexdigest() for f, v in ORIG.items()}
BASE_FILES = set(os.listdir(PKG))

def restore():
    for f, v in ORIG.items():
        open(f, "w").write(v)
    for f in ORIG:
        assert hashlib.sha256(open(f).read().encode()).hexdigest() == SUMS[f], f
    for extra in set(os.listdir(PKG)) - BASE_FILES:
        os.remove(os.path.join(PKG, extra))
    assert set(os.listdir(PKG)) == BASE_FILES

def apply(edits, newfiles):
    for f, old, new in edits:
        s = open(f).read()
        if old not in s:
            restore(); return "ANCHOR-MISSING:%s" % old[:60]
        open(f, "w").write(s.replace(old, new, 1))
    for name, body in newfiles:
        open(os.path.join(PKG, name), "w").write(body)
    return None

def run(pkgs, testflags=None):
    cmd = ["go", "test"] + pkgs + ["-count=1"] + (testflags or [])
    r = subprocess.run(cmd, capture_output=True, text=True)
    return r.returncode, r.stdout + r.stderr

def probe(pid, desc, edits, newfiles=(), pkgs=None, expect="RED"):
    restore()
    err = apply(edits, newfiles)
    if err:
        restore()
        print(f"{pid}\t{err}\t{desc}")
        return {"id": pid, "desc": desc, "result": err, "expect": expect}
    code, out = run(pkgs or ["./internal/terminalbackend/"])
    fails = sorted({l.split()[2] for l in out.splitlines()
                    if l.strip().startswith("--- FAIL:") and len(l.split()) > 2})
    verdict = "RED" if code != 0 else "GREEN"
    restore()
    status = "OK" if verdict == expect else ("*** UNEXPECTED ***")
    print(f"{pid}\t{verdict}\t(expect {expect}) {status}\t{desc}")
    if fails:
        print("      failing:", ", ".join(fails[:8]))
    elif verdict == "RED":
        print("      (no --- FAIL lines; build error or t.Fatal at package level)")
        print("      " + "\n      ".join([l for l in out.splitlines() if "FAIL" in l or ".go:" in l][:6]))
    return {"id": pid, "desc": desc, "result": verdict, "expect": expect, "failing": fails}
